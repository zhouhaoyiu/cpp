cmake_minimum_required(VERSION 3.16...4.4)
project(javascript-interpreter LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

add_library(interpreter src/Lexer.cpp src/Parser.cpp src/Interpreter.cpp)
target_include_directories(interpreter PUBLIC include)
add_executable(node src/main.cpp)
add_executable(lexer src/test-lexer.cpp)
add_executable(parser src/test-parser.cpp)
foreach(target node lexer parser)
    target_link_libraries(${target} PRIVATE interpreter)
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(regression tests/regression.cpp)
    target_link_libraries(regression PRIVATE interpreter)
    add_test(NAME lexer-parser-regressions COMMAND regression)
    add_test(NAME command-line-regressions COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
        $<TARGET_FILE:node> $<TARGET_FILE:lexer> $<TARGET_FILE:parser>)
    set_tests_properties(lexer-parser-regressions command-line-regressions PROPERTIES TIMEOUT 30)
endif()
